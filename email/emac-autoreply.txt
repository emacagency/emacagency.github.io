// EMAC auto-reply for website form messages.
// Runs inside agencyemac@gmail.com and sends the designed thank-you email
// to whoever filled the form on emacagency.github.io.
// Each form message is answered once, even when Gmail groups them in one conversation.

// Press Run once on any function: old messages are skipped,
// and a check runs every minute from then on.
function setup() {
  ScriptApp.getProjectTriggers().forEach(function (t) { ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('emacAutoReply').timeBased().everyMinutes(1).create();
  run(true);
}

var LABEL = 'emac-replied';
var PAGE = 'https://emacagency.github.io/email/autoreply.html';
var FORM_SUBJECT = 'طلب حملة جديد من موقع إي ماك';
var SUBJECT = 'شكرًا لتواصلك مع إي ماك';
var TEXT = 'شكرًا جزيلًا لتواصلك مع إي ماك! وصلتنا رسالتك بنجاح، وسيرد عليك فريقنا خلال 48 ساعة.\nhttps://emacagency.github.io';

function emacAutoReply() {
  if (!ScriptApp.getProjectTriggers().length) return setup();
  run(false);
}

function run(markOnly) {
  var props = PropertiesService.getScriptProperties();
  var saved = props.getProperty('done');
  if (saved === null) markOnly = true;
  var done = JSON.parse(saved || '{}');
  var now = Date.now();
  var instant = JSON.parse(props.getProperty('instant') || '{}');
  var label = GmailApp.getUserLabelByName(LABEL) || GmailApp.createLabel(LABEL);
  var threads = GmailApp.search('from:formsubmit.co subject:"' + FORM_SUBJECT + '" newer_than:2d');
  threads.forEach(function (t) {
    t.getMessages().forEach(function (m) {
      var id = m.getId();
      if (done[id]) return;
      done[id] = now;
      if (markOnly) return;
      if (!/formsubmit/i.test(m.getFrom()) || m.getSubject().indexOf(FORM_SUBJECT) < 0) return;
      var to = visitorEmail(m);
      if (!to) return;
      if (now - (instant[to.toLowerCase()] || 0) > 30 * 60e3) sendReply(to);
      t.addLabel(label);
    });
  });
  for (var k in done) if (now - done[k] > 4 * 864e5) delete done[k];
  props.setProperty('done', JSON.stringify(done));
}

function sendReply(to) {
  var html = UrlFetchApp.fetch(PAGE).getContentText();
  GmailApp.sendEmail(to, SUBJECT, TEXT, { htmlBody: html, name: 'EMAC | إي ماك' });
}

// The website sends the form straight here (Deploy > New deployment > Web app,
// execute as Me, access: Anyone). The request lands in EMAC's inbox and the
// visitor gets the thank-you right away, with no outside form service.
var FIELDS = [['Name', 'الاسم'], ['Organization', 'الجهة'], ['email', 'البريد'], ['Research type', 'نوع البحث'], ['Message', 'الرسالة']];

function doPost(e) {
  var p = (e && e.parameter) || {};
  var to = String(p.email || '').trim();
  if (p._honey || !String(p.Name || '').trim() || !/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(to)) return reply({ ok: false });
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var props = PropertiesService.getScriptProperties();
    var log = JSON.parse(props.getProperty('instant') || '{}'), now = Date.now(), today = 0;
    for (var k in log) { if (now - log[k] > 864e5) delete log[k]; else today++; }
    if (today >= 80) return reply({ ok: false, limit: true });
    var rows = FIELDS.map(function (f) {
      var v = String(p[f[0]] || '').replace(/[<>&]/g, function (c) { return { '<': '&lt;', '>': '&gt;', '&': '&amp;' }[c]; });
      return '<tr><td style="padding:8px 12px;background:#F3F4F6;font-weight:bold">' + f[1] + '</td><td style="padding:8px 12px;white-space:pre-wrap">' + v + '</td></tr>';
    }).join('');
    var plain = FIELDS.map(function (f) { return f[1] + ': ' + (p[f[0]] || ''); }).join('\n');
    GmailApp.sendEmail(Session.getEffectiveUser().getEmail(), FORM_SUBJECT, plain, {
      htmlBody: '<div dir="rtl" style="font-family:Tahoma,Arial"><table style="border-collapse:collapse">' + rows + '</table></div>',
      replyTo: to, name: 'موقع إي ماك'
    });
    var key = to.toLowerCase();
    if (!log[key] || now - log[key] > 10 * 60e3) { sendReply(to); log[key] = now; }
    props.setProperty('instant', JSON.stringify(log));
  } finally { lock.releaseLock(); }
  return reply({ ok: true });
}

function reply(o) {
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}

function visitorEmail(m) {
  var ok = function (a) { return a && !/formsubmit|agencyemac@gmail\.com|noreply|no-reply/i.test(a); };
  var r = (m.getReplyTo() || '').match(/[\w.+-]+@[\w-]+(\.[\w-]+)+/);
  if (r && ok(r[0])) return r[0];
  var all = m.getPlainBody().match(/[\w.+-]+@[\w-]+(\.[\w-]+)+/g) || [];
  for (var i = 0; i < all.length; i++) if (ok(all[i])) return all[i];
  return null;
}
